Проблема: Node.js скрипт не находит @prisma/client
Решение: Используем прямой SQL запрос

──────────────────────────────────────────────────────────────
📋 ВЫПОЛНИТЕ НА СЕРВЕРЕ (скопируйте целиком):
──────────────────────────────────────────────────────────────

cd /var/www/newfatiha

# Шаг 1: Пометить миграцию как примененную
npx prisma migrate resolve --applied 20260320000000_add_teacher_enrollment_system

# Шаг 2: Генерация Prisma Client
npx prisma generate

# Шаг 3: Создать администратора через SQL
sudo -u postgres psql -d fatiha << 'EOF'
-- Удалить старого админа
DELETE FROM "User" WHERE email = 'admin@fatiha.ru';

-- Создать нового админа
-- Пароль: admin123 (bcrypt hash)
INSERT INTO "User" (
  id,
  email,
  password,
  name,
  role,
  "emailVerified",
  status,
  gender,
  "createdAt",
  "updatedAt"
) VALUES (
  gen_random_uuid(),
  'admin@fatiha.ru',
  '$2a$10$N9qo8uLOickgx2ZMRZoMye.IjzKrMa3s83si9GeCAos99JxHm8jqW',
  'Администратор',
  'ADMIN',
  true,
  'ACTIVE',
  'NOT_SPECIFIED',
  NOW(),
  NOW()
);

-- Проверить что создан
SELECT email, name, role, status FROM "User" WHERE email = 'admin@fatiha.ru';
EOF

# Шаг 4: Перезапустить приложение
pm2 restart fatiha

echo ""
echo "✅ ГОТОВО!"
echo "🌐 Откройте: https://fatiha.ru"
echo "📧 Email: admin@fatiha.ru"
echo "🔑 Password: admin123"
