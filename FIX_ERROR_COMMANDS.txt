📋 ВЫПОЛНИТЕ ЭТИ КОМАНДЫ НА СЕРВЕРЕ:

──────────────────────────────────────────────────────────────
ШАГ 1: Применить миграции
──────────────────────────────────────────────────────────────
cd /var/www/newfatiha
npx prisma migrate deploy


──────────────────────────────────────────────────────────────
ШАГ 2: Проверить, что поля добавлены
──────────────────────────────────────────────────────────────
sudo -u postgres psql -d fatiha -c "\d \"User\"" | grep emailVerified

# Должна быть строка:
# emailVerified | boolean | | not null | false


──────────────────────────────────────────────────────────────
ШАГ 3: Создать администратора (ПОСЛЕ миграций!)
──────────────────────────────────────────────────────────────
cat > /tmp/create-admin.js << 'EOF'
const { PrismaClient } = require('@prisma/client');
const bcrypt = require('bcryptjs');
const prisma = new PrismaClient();

(async () => {
  try {
    const hash = await bcrypt.hash('admin123', 10);
    await prisma.user.deleteMany({ where: { email: 'admin@fatiha.ru' } });
    const user = await prisma.user.create({
      data: {
        email: 'admin@fatiha.ru',
        password: hash,
        name: 'Администратор',
        role: 'ADMIN',
        emailVerified: true,
        status: 'ACTIVE',
        gender: 'NOT_SPECIFIED'
      }
    });
    console.log('✅ Администратор создан:', user.email);
  } catch (e) {
    console.error('❌ Ошибка:', e.message);
    process.exit(1);
  } finally {
    await prisma.$disconnect();
  }
})();
EOF

node /tmp/create-admin.js


──────────────────────────────────────────────────────────────
ШАГ 4: Перезапустить приложение
──────────────────────────────────────────────────────────────
pm2 restart fatiha


──────────────────────────────────────────────────────────────
ШАГ 5: Проверить логи
──────────────────────────────────────────────────────────────
pm2 logs fatiha --lines 50

ШАГ 6: Войти в систему
