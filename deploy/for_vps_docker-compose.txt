services:
  db:
    image: postgres:16-alpine
    container_name: fatiha_postgres
    restart: always
    environment:
      POSTGRES_USER: fatiha_user
      POSTGRES_PASSWORD: change_me_db_password
      POSTGRES_DB: fatiha_db
    volumes:
      - fatiha_pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U fatiha_user -d fatiha_db"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - fatiha_network

  app:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: fatiha_app
    restart: always
    env_file:
      - .env.production
    environment:
      DATABASE_URL: postgresql://fatiha_user:change_me_db_password@db:5432/fatiha_db?connection_limit=20
    ports:
      - "3000:3000"
    depends_on:
      db:
        condition: service_healthy
    networks:
      - fatiha_network

volumes:
  fatiha_pgdata:

networks:
  fatiha_network:
    driver: bridge
