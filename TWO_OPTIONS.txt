──────────────────────────────────────────────────────────────

# 1. Удалить БД
sudo -u postgres psql -c "DROP DATABASE fatiha;"
sudo -u postgres psql -c "CREATE DATABASE fatiha;"
sudo -u postgres psql -c "GRANT ALL PRIVILEGES ON DATABASE fatiha TO fatiha_user;"

# 2. Применить миграции
cd /var/www/newfatiha
npx prisma migrate deploy

# 3. Создать админа
npx prisma db seed

# 4. Пересобрать приложение
rm -rf .next node_modules
npm install
npm run build

# 5. Перезапустить
pm2 restart fatiha
